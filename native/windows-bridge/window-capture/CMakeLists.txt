cmake_minimum_required(VERSION 3.24)
project(CrossWarpWindowCapture LANGUAGES CXX)
add_library(crosswarp-window-capture STATIC window_capture.cpp)
target_compile_features(crosswarp-window-capture PRIVATE cxx_std_20)
target_compile_definitions(crosswarp-window-capture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(crosswarp-window-capture PUBLIC .)
target_link_libraries(crosswarp-window-capture PUBLIC d3d11 dxgi windowsapp)
add_executable(crosswarp-window-capture-smoke capture_smoke.cpp)
target_compile_features(crosswarp-window-capture-smoke PRIVATE cxx_std_20)
target_compile_definitions(crosswarp-window-capture-smoke PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(crosswarp-window-capture-smoke PRIVATE crosswarp-window-capture)
